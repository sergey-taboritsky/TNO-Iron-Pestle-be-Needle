defined_text = {
	name = TNOIPN_SYR_GetParliamentLoyalty
	text = {
		trigger = {
			check_variable = { TNOIPN_parliament_loyalty^v > 69 }
		}
		localization_key = TNOIPN_SYR_loyalty_very_good
	}
	text = {
		trigger = {
			check_variable = { TNOIPN_parliament_loyalty^v > 50 }
			check_variable = { TNOIPN_parliament_loyalty^v < 70 }
		}
		localization_key = TNOIPN_SYR_loyalty_good
	}
	text = {
		trigger = {
			check_variable = { TNOIPN_parliament_loyalty^v = 50 }
		}
		localization_key = TNOIPN_SYR_loyalty_weak
	}
	text = {
		trigger = {
			check_variable = { TNOIPN_parliament_loyalty^v > 30 }
			check_variable = { TNOIPN_parliament_loyalty^v < 50 }
		}
		localization_key = TNOIPN_SYR_loyalty_opposition
	}
	text = {
		trigger = {
			check_variable = { TNOIPN_parliament_loyalty^v < 31 }
		}
		localization_key = TNOIPN_SYR_loyalty_strong_opposition
	}
	text = {
		localization_key = "Error"
	}
}
